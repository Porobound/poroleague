# Admin Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Organizer tools for a running season: result edits and entries, flag resolution, drops, weekly pairings post, end of season (tiebreak ranks, finalize), season-deletion post.

**Architecture:** Pure pieces stay testable: tie clusters in `standings.ts`, week rounds in `time.ts`, every Discord payload in `discord.ts` (Vitest). DB writes live in `src/lib/server/{matches,seasons}.ts`, each with its audit row in the same transaction. One new admin page (`/seasons/[slug]/admin/matches`) holds every result edit; the season admin page gains a play/end-of-season block; the players page gains drops. Server-rendered forms only, no client JS beyond the existing `ConfirmButton` dialog.

**Tech Stack:** SvelteKit (Svelte 5 runes), Drizzle + better-sqlite3, Vitest.

**Design boards** (canvas in `design-artifacts` memory): `Admin` (flags, pairings), `AdminMatches`, `AdminPlayers` (in play), `AdminEndSeason`. Extract, don't paste: reuse `pb-` classes and `src/app.css` helpers.

**Decisions (made in planning, owner can overrule):**

- Organizer edit on a flagged result resolves the flag (`flagResolvedBy/At`). Edits of unflagged results and organizer-entered results stay flaggable by players: the organizer can be wrong too, and it needs no new state.
- Organizer-entered results need legends and dice like a player report (same `parseReport`); the dialog pre-fills last-used legends.
- An organizer-entered result for an unreported match posts to the results channel (it is a result); edits, keeps, drops and season deletions post to the organizer channel.
- Pairings: one Discord message per round (stays under the 2000-character limit at any `roundsPerWeek`), plain `content` with `<@id>` mentions, `allowed_mentions.users` = exactly the paired players. Posts the current week (week 1 before the start date). Hidden once reporting is closed.
- End of season opens when the deadline has passed **or** every non-void match is reported. Finalize needs zero open flags and zero unresolved ties. Finalizing is final (status `finished`, nothing editable).
- Tie ordering: ↑/↓ buttons post the whole new order; any posted order sets `tiebreak` 1..n for that cluster. "Keep this order" posts the current order. Clusters are computed ignoring organizer ranks, so a settled tie can still be reordered.
- Tiebreak and finalize write audit rows (`tiebreak`, `finalize`); the public audit page shows them.

**Deliberately skipped:** undo drop; top-cut qualifier marks; inline unplayed list on the season page (links to the Matches filter instead); "recent audit" table on the admin page (public audit page exists); mobile admin (issue #18).

**Env:** new `WEBHOOK_PAIRINGS_URL` (may be empty). Every env var must be present or the app refuses to start, so `.env` needs the line before the server reloads.

export function roundsOfWeek(week: number, roundsPerWeek: number, maxRound: number) {

```ts
export type WebhookBody = {
	content?: string;
	embeds?: Embed[];
	allowed_mentions: { parse: []; users?: string[] };
};

export function organizerEditEmbed(
	m: MatchPost,
	by: string,
	o: { resolves: boolean; kept: boolean }
): Embed {
	return {
		title: `${o.resolves ? 'Flag resolved' : 'Result changed'} by ${escapeMd(by)}`,
		...base(m),
		description: `${o.kept ? 'Kept as reported' : 'New result'}: ${score(m)} (${m.report.p1Legend} vs ${m.report.p2Legend})`,
		color: SIGNAL
	};
}

export const dropEmbed = (o: {
	player: string;
	by: string;
	where: string;
	url: string;
}): Embed => ({
	title: `${escapeMd(o.player)} dropped`,
	url: o.url,
	description: `By ${escapeMd(o.by)}. All their matches are void.`,
	color: SIGNAL,
	footer: { text: o.where },
	timestamp: new Date().toISOString()
});

export const seasonDeletedEmbed = (name: string, by: string): Embed => ({
	title: `Season deleted: ${escapeMd(name)}`,
	description: `By ${escapeMd(by)}. It's hidden from the site; the data is kept.`,
	color: SIGNAL,
	footer: { text: 'Poroleague admin' },
	timestamp: new Date().toISOString()
});

export type Pairing = { group: number | null; p1Id: string; p2Id: string };

// Mentions only ping from `content`, and only the users listed, so nothing else can be pinged.
export function pairingsPost(title: string, round: number, pairs: Pairing[]): WebhookBody {
	const lines = pairs.map(
		(p) => `${p.group ? `${groupName(p.group)}: ` : ''}<@${p.p1Id}> vs <@${p.p2Id}>`
	);
	return {
		content: [`**${escapeMd(title)} · Round ${round}**`, ...lines].join('\n'),
		allowed_mentions: { parse: [], users: [...new Set(pairs.flatMap((p) => [p.p1Id, p.p2Id]))] }
	};
}

import {
	WEBHOOK_ORGANIZERS_URL,
	WEBHOOK_PAIRINGS_URL,
	WEBHOOK_RESULTS_URL
} from '$app/env/private';
import type { Embed, WebhookBody } from '#lib/discord.ts';

// Fire-and-forget: failures are logged and never undo or block the change that triggered them.
async function post(url: string, body: WebhookBody) {
	if (!url) return;
	try {
		const res = await fetch(url, {
			method: 'POST',
			headers: { 'content-type': 'application/json' },
			body: JSON.stringify(body)
		});
		if (!res.ok) console.error(`Discord webhook failed: ${res.status}`);
	} catch (e) {
		console.error(e);
	}
}

// Names and flag reasons are user text: embeds never ping @everyone, roles or users.
const embed = (e: Embed): WebhookBody => ({ embeds: [e], allowed_mentions: { parse: [] } });

export const postResult = (e: Embed) => void post(WEBHOOK_RESULTS_URL, embed(e));
export const postToOrganizers = (e: Embed) => void post(WEBHOOK_ORGANIZERS_URL, embed(e));
export const pairingsEnabled = () => !!WEBHOOK_PAIRINGS_URL;
// In order, one message per round.
export const postPairings = (bodies: WebhookBody[]) =>
	void (async () => {
		for (const b of bodies) await post(WEBHOOK_PAIRINGS_URL, b);
	})();

	m.reportedBy

// Organizer fix or entry. Resolves an open flag; an unreported match gets the organizer as reporter.
export function organizerEdit(m: Match, actorId: string, values: Report) {
	const now = new Date();
	db.transaction((tx) => {
		tx.update(matches)
			.set({
				...values,
				...(m.reportedBy ? {} : { reportedBy: actorId, reportedAt: now }),
				...(m.flaggedBy && !m.flagResolvedAt
					? { flagResolvedBy: actorId, flagResolvedAt: now }
					: {})
			})
			.where(eq(matches.id, m.id))
			.run();
		tx.insert(audit)
			.values({
				seasonId: m.seasonId,
				matchId: m.id,
				actorId,
				action: 'edit',
				before: resultOf(m),
				after: values
			})
			.run();
	});
}

export function keepResult(m: Match, actorId: string) {

		tx.update(matches)
			.set({ flagResolvedBy: actorId, flagResolvedAt: new Date() })

			.run();

			.values({
				seasonId: m.seasonId,
				matchId: m.id,
				actorId,
				action: 'resolve',
				after: resultOf(m)
			})

}

export function setTiebreaks(seasonId: number, order: string[], actorId: string) {
	db.transaction((tx) => {
		order.forEach((playerId, i) =>
			tx
				.update(entries)
				.set({ tiebreak: i + 1 })
				.where(and(eq(entries.seasonId, seasonId), eq(entries.playerId, playerId)))
				.run()
		);
		tx.insert(audit).values({ seasonId, actorId, action: 'tiebreak', after: { order } }).run();
	});
}

export function finalizeSeason(seasonId: number, actorId: string) {
	db.transaction((tx) => {
		tx.update(seasons).set({ status: 'finished' }).where(eq(seasons.id, seasonId)).run();
		tx.insert(audit).values({ seasonId, actorId, action: 'finalize' }).run();
	});
}
```
